local RS = game:GetService("ReplicatedStorage")

local lp = Players.LocalPlayer

local fishNames = {}
local fishVFX = RS:WaitForChild("FishVFX")
for _, fish in pairs(fishVFX:GetChildren()) do
    table.insert(fishNames, fish.Name)
end
table.sort(fishNames)

local spawnList = {}
for _, sp in pairs(spawnPoints:GetChildren()) do
    table.insert(spawnList, sp.Name)
end
table.sort(spawnList)

local function teleportTo(name)
    local sp = spawnPoints:FindFirstChild(name)
    if sp then
        local char2 = lp.Character
        if char2 and char2:FindFirstChild("HumanoidRootPart") then
            if sp:IsA("BasePart") then
                char2.HumanoidRootPart.CFrame = sp.CFrame + Vector3.new(0, 3, 0)
            elseif sp:IsA("Model") then
                char2.HumanoidRootPart.CFrame = sp:GetModelCFrame() + Vector3.new(0, 3, 0)
            else
                local part = sp:FindFirstChildWhichIsA("BasePart")
                if part then
                    char2.HumanoidRootPart.CFrame = part.CFrame + Vector3.new(0, 3, 0)
                end
            end
        end
    end
end

local function castAndConfirm()
    local timeout = 0
    repeat
        pcall(function()
            Events:WaitForChild("Fishing"):FireServer()
        end)
        task.wait(0.5)
        timeout = timeout + 0.5
        if timeout >= 10 then
            return false
        end
    until (char:GetAttribute("Fishing") or not autoFish)
    return true
end

local function waitForFishDeath()
    local timeout = 0
    repeat
        task.wait(0.1)
        timeout = timeout + 0.1
        if timeout >= 10 then return end
    until lp:GetAttribute("FishID")

    local fishID = lp:GetAttribute("FishID")
    local fish = workspace:WaitForChild("Fishes"):FindFirstChild(tostring(fishID))
    if not fish then return end

    timeout = 0
    repeat
        task.wait(0.1)
        timeout = timeout + 0.1
        if timeout >= 60 then return end
        if not fish or not fish.Parent then return end
    until fish.Value <= 0
end

local function abandonFish()
    if abandonMethod == "recast" then
        pcall(function()
            Events:WaitForChild("Fishing"):FireServer()
        end)
    elseif abandonMethod == "reequip" then
        pcall(function()
            Events:WaitForChild("ToggleHotbar"):InvokeServer("1")
            task.wait(0.2)
            Events:WaitForChild("ToggleHotbar"):InvokeServer("1")
        end)
    end
end

Events.FishingMinigame.OnClientEvent:Connect(function(p1)
    if targetFishEnabled then
        local fishName = p1.FishName or "Fish"
        if not targetFishList[fishName] then
            task.wait(0.2)
            abandonFish()
        end
    end
end)

FishTab:CreateToggle({
    Name = "Auto Fish",
    CurrentValue = false,
    Flag = "AutoFish",
    Callback = function(val)
        autoFish = val
        if val then
            task.spawn(function()
                while autoFish do
                    pcall(function()
                        local success = castAndConfirm()
                        if success then
                            waitForFishDeath()
                        end
                    end)
                    task.wait(0.5)
                end
            end)
        end
    end
})

    char = lp.Character or char
    if barLock then
        pcall(function()
            local fishingUI = MainGui.Fishing
            if fishingUI.Visible then
                local bar = fishingUI.BarFrame.Bar
                bar.Position = UDim2.new(0.8, 0, bar.Position.Y.Scale, 0)
                RS.Fishing:FireServer("1")
            end
        end)
    end
end)
